import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {pathToFileURL} from 'node:url';import path from 'node:path';
const base='benchmarks/language/results/pr73-luna-final';const read=f=>JSON.parse(fs.readFileSync(base+'/'+f,'utf8'));const sha=x=>createHash('sha256').update(x).digest('hex');const write=(f,x)=>fs.writeFileSync(base+'/'+f,JSON.stringify(x,null,2)+'\n',{flag:'wx'});
const {languageReviewPacket,adjudicateLanguageGate}=await import(pathToFileURL(path.resolve('packages/core/dist/bench/language-review.js')));
const reports=['development','held-out'].map(s=>{const completion=read(s+'/completion.json');assert.equal(completion.completed,22);assert.equal(completion.expected,22);assert.equal(completion.builtRuntimeUnchanged,true);return read(s+'/report.json');});
const observations=new Map(reports.flatMap(r=>r.cases.map(c=>[`${c.id}/${c.run}`,c])));
const packet=languageReviewPacket(reports,read('input-review.json'));const packetCases=new Map(packet.cases.map(c=>[`${c.id}/${c.run}`,c]));
const packets=fs.readdirSync(base+'/reviews').filter(f=>/^batch-\d+-packet\.json$/.test(f)).sort();const cases=[],contexts=[],provenance=[];let reviewer;
for(const f of packets){const p=read('reviews/'+f);const r=read('reviews/'+f.replace('-packet','-review'));const x=read('reviews/'+f.replace('-packet','-context'));const {packetFingerprint,...body}=p;assert.equal(sha(JSON.stringify(body)),packetFingerprint);assert.equal(r.packetFingerprint,packetFingerprint);assert.equal(x.packetFingerprint,packetFingerprint);
 for(const c of p.cases)assert.deepEqual(c,packetCases.get(`${c.id}/${c.run}`));
 const expected=p.cases.map(c=>`${c.id}/${c.run}`).sort();assert.deepEqual(r.cases.map(c=>`${c.id}/${c.run}`).sort(),expected);assert.deepEqual(x.cases.map(c=>`${c.id}/${c.run}`).sort(),expected);
 assert.deepEqual(x.reviewer,r.reviewer);
 for(const extra of p.extra){const raw=observations.get(`${extra.id}/${extra.run}`);assert.deepEqual(extra.sourceArchive,raw.sourceArchive);const expectedContexts=[...new Map(raw.queries.map(q=>[`${q.queryLanguage}/${q.explicitView}`,{queryLanguage:q.queryLanguage,explicitView:q.explicitView,evidence:q.reviewContext,knowledgeLanguage:q.contextKnowledgeLanguage}])).values()];assert.deepEqual(extra.contexts,expectedContexts);const grade=x.cases.find(c=>c.id===extra.id&&c.run===extra.run);assert.equal(grade.contexts.length,4);assert.deepEqual(grade.contexts.map(c=>`${c.queryLanguage}/${c.explicitView}`).sort(),expectedContexts.map(c=>`${c.queryLanguage}/${c.explicitView}`).sort());for(const c of grade.contexts){for(const k of ['usefulContext','sourceEntailed','qualificationPreserved','languageFramingCompliant','unsafeFactualPromotion','safeExclusion'])assert.equal(typeof c[k],'boolean');assert(c.reason.length);if(c.usefulContext)assert(expectedContexts.find(e=>e.queryLanguage===c.queryLanguage&&e.explicitView===c.explicitView).evidence.length>0);}}
 if(reviewer)assert.deepEqual(r.reviewer,reviewer);else reviewer=r.reviewer;
 cases.push(...r.cases);contexts.push(...x.cases);provenance.push({batch:f,packetFingerprint,reviewJsonSha256:sha(JSON.stringify(r)),contextReviewJsonSha256:sha(JSON.stringify(x)),canonicalCasesMatched:true});
}
assert.equal(cases.length,44);assert.equal(new Set(cases.map(c=>`${c.id}/${c.run}`)).size,44);
const review={schemaVersion:'language-output-review-v2',packetFingerprint:packet.packetFingerprint,reviewer,cases};const gate=adjudicateLanguageGate(reports,read('input-review.json'),review);
write('review-packet.json',packet);write('output-review.json',review);write('context-review.json',{schemaVersion:'language-context-review-v1',packetFingerprint:packet.packetFingerprint,reviewer,cases:contexts});write('review-provenance.json',provenance);write('gate.json',gate);
console.log(JSON.stringify({gatePassed:gate.releaseEligible,failures:gate.failures,groups:gate.groups.map(g=>({split:g.split,run:g.run,metrics:g.metrics}))},null,2));
