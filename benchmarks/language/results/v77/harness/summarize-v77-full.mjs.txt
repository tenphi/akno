import fs from 'node:fs';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';
const read=p=>JSON.parse(fs.readFileSync(p,'utf8'));const hash=p=>createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const original=read('tmp/v77-full-original-gate.json'),completion=read('tmp/v77-full-completion-gate.json'),review=read('tmp/language-v77-full-output-review.json');
assert.equal(original.outputReviewFingerprint,completion.outputReviewFingerprint);
const reports=['development','held-out'].map(s=>read(`bench-results/language-full-${s}-v77.json`));
const groups=completion.groups.map(g=>{
 const report=reports.find(r=>r.split===g.split),cases=report.cases.filter(c=>c.run===g.run),writable=cases.filter(c=>c.admission==='writable'),readOnly=cases.filter(c=>c.admission==='read-only');
 const answers=writable.flatMap(c=>c.queries),controls=readOnly.flatMap(c=>c.queries);
 assert.equal(answers.length,80);assert.equal(controls.length,8);
 return {split:g.split,run:g.run,...g.metrics,publishedWritableAnswers:answers.filter(q=>q.reviewAnswer!==null).length,nullWritableAnswers:answers.filter(q=>q.reviewAnswer===null).length,readOnlyObservations:controls.length,readOnlyPublishedAnswers:controls.filter(q=>q.reviewAnswer!==null).length,sourceBytesVerified:cases.every(c=>c.bytesStable===true),replayVerified:cases.every(c=>c.replayOutcome==='replayed'),availabilityCases:cases.filter(c=>c.availabilityFailure).map(c=>c.id)};
});
const total=field=>groups.reduce((n,g)=>n+g[field],0);
const summary={runtimeCommit:completion.provenance.runtimeCommit,corpusFingerprint:completion.provenance.corpusFingerprint,originalGatePassed:original.releaseEligible,completionGatePassed:completion.completionEligible,originalFailures:original.failures,completionFailures:completion.failures,groups,totals:{writableObservations:320,readOnlyObservations:32,usefulWritableAnswers:groups.reduce((n,g)=>n+g.usefulQualifiedAnswerCoverage.numerator,0),publishedWritableAnswers:total('publishedWritableAnswers'),nullWritableAnswers:total('nullWritableAnswers'),completeRetainedSets:groups.reduce((n,g)=>n+g.usefulRetentionCoverage.numerator,0),usefulRetrievals:groups.reduce((n,g)=>n+g.qualifiedRetrievalCoverage.numerator,0),unsupportedRetainedOutputs:total('unsupportedRetainedOutputs'),unsupportedNonnullAnswers:total('unsupportedNonnullAnswers'),qualificationErrors:total('translationQualificationErrors'),languageErrors:total('acceptedLanguageViolations'),promotions:total('unsafeFactualPromotion'),sourceByteErrors:total('sourceByteChanges'),availabilityCases:groups.reduce((n,g)=>n+g.availabilityCases.length,0)},reviewSha256:hash('tmp/language-v77-full-output-review.json'),originalGateSha256:hash('tmp/v77-full-original-gate.json'),completionGateSha256:hash('tmp/v77-full-completion-gate.json'),notAUniversalReliabilityClaim:true};
fs.writeFileSync('tmp/v77-full-output-summary.json',JSON.stringify(summary,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(summary));
