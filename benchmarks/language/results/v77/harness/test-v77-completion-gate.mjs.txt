import assert from 'node:assert/strict';import fs from 'node:fs';import {completionGate} from './v77-completion-gate.mjs';
const contract=JSON.parse(fs.readFileSync('tmp/v77-completion-contract.json','utf8'));
const base={schemaVersion:'language-quality-gate-v2',independentlyReviewed:true,releaseEligible:false,scope:'Synthetic arithmetic control only',thresholds:{...contract.completionThresholds,usefulQualifiedAnswerCoverage:0.9},failures:['development/run-1:usefulQualifiedAnswerCoverage'],groups:['development','held-out'].flatMap(split=>[1,2].map(run=>({split,run,metrics:{usefulRetentionCoverage:{numerator:8,denominator:10,rate:0.8},qualifiedRetrievalCoverage:{numerator:32,denominator:40,rate:0.8},usefulQualifiedAnswerCoverage:{numerator:64,denominator:80,rate:0.8}}})))};
assert(completionGate(base,contract).completionEligible);assert.equal(base.thresholds.usefulQualifiedAnswerCoverage,0.9);assert.equal(base.failures.length,1);
const low=structuredClone(base);low.groups[0].metrics.usefulQualifiedAnswerCoverage={numerator:63,denominator:80,rate:63/80};assert(!completionGate(low,contract).completionEligible);
for(const failure of ['unsupportedRetainedOutputs','unsupportedNonnullAnswers','translationQualificationErrors','acceptedLanguageViolations','unsafeFactualPromotion','sourceByteChanges','availability','admission','ordinaryProse','usefulRetentionCoverage','qualifiedRetrievalCoverage']){const bad=structuredClone(base);bad.failures.push('held-out/run-2:'+failure);assert(completionGate(bad,contract).failures.includes('held-out/run-2:'+failure));assert(!completionGate(bad,contract).completionEligible);}
const pooled=structuredClone(low);for(const g of pooled.groups.slice(1))g.metrics.usefulQualifiedAnswerCoverage={numerator:80,denominator:80,rate:1};assert(!completionGate(pooled,contract).completionEligible);
const tampered=structuredClone(contract);tampered.completionThresholds.unsupportedNonnullAnswers=1;assert.throws(()=>completionGate(base,tampered));
const wrongDenom=structuredClone(base);wrongDenom.groups[0].metrics.usefulQualifiedAnswerCoverage.denominator=88;assert.throws(()=>completionGate(wrongDenom,contract));
console.log(JSON.stringify({passed:true,providerCalls:0,controls:16,syntheticArithmeticOnly:true}));
